Understanding Middleware in Express.js

Middleware functions are functions that have access to the request object (req), the response object (res), and the next middleware function in the application's request-response cycle. They can execute code, modify the request/response objects, end the request-response cycle, or call the next middleware.

🟣 Purple Theme: Middleware is the backbone of Express applications – it powers everything from authentication to logging to error handling.

🔄 Middleware Flow:

Request → Middleware 1 → Middleware 2 → Middleware 3 → Route Handler → Response

Each middleware can process the request, then call next() to pass control to the next middleware.

const express = require('express');
const app = express();

// Simple middleware example
app.use((req, res, next) => {
  console.log('Time:', Date.now());
  next(); // Pass control to the next middleware
});

app.get('/', (req, res) => {
  res.send('Hello World!');
});

app.listen(3000);

Application-Level Middleware

Application-level middleware are bound to an instance of app using app.use() or app.METHOD(). They run for every request or for specific routes.

const app = express();

// Logger middleware - runs for every request
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next();
});

// Middleware for specific path
app.use('/api', (req, res, next) => {
  console.log('API request received');
  next();
});

// Middleware for specific HTTP method and path
app.get('/protected', (req, res, next) => {
  console.log('Protected route accessed');
  next();
}, (req, res) => {
  res.send('Protected data');
});

Built-in Middleware in Express

Express comes with several built-in middleware functions that handle common tasks.

MiddlewarePurpose
express.json()Parses JSON request bodies
express.urlencoded({ extended: true })Parses URL-encoded form data
express.static()Serves static files (CSS, images, etc.)
express.Router()Creates modular route handlers
const express = require('express');
const app = express();

// Parse JSON bodies
app.use(express.json());

// Parse URL-encoded bodies
app.use(express.urlencoded({ extended: true }));

// Serve static files from 'public' folder
app.use(express.static('public'));

app.post('/api/data', (req, res) => {
  console.log(req.body); // Access parsed data
  res.json({ received: req.body });
});
✅ Without express.json(), req.body would be undefined.

Custom Middleware

You can create your own middleware functions to handle specific tasks like authentication, logging, validation, etc.

// Authentication middleware
const authenticate = (req, res, next) => {
  const token = req.headers['authorization'];
  if (!token) {
    return res.status(401).json({ error: 'Unauthorized' });
  }
  // Verify token logic here
  req.user = { id: 1, name: 'John' }; // Attach user to request
  next();
};

// Logging middleware
const logger = (req, res, next) => {
  const start = Date.now();
  res.on('finish', () => {
    const duration = Date.now() - start;
    console.log(`${req.method} ${req.url} - ${res.statusCode} - ${duration}ms`);
  });
  next();
};

// Rate limiting middleware
const rateLimiter = (req, res, next) => {
  const ip = req.ip;
  // Implementation would track requests per IP
  next();
};

app.use(logger);
app.use(rateLimiter);
app.use('/api/admin', authenticate); // Only apply to admin routes

Route-Level Middleware in Express.js

Route-level middleware is applied to specific routes or route groups. It can be used for route-specific validation, authentication, or data processing.

🟣 Route-level middleware is more granular than application-level middleware and is ideal for route-specific logic.
const express = require('express');
const app = express();

// Middleware function
const validateUser = (req, res, next) => {
  const { id } = req.params;
  if (!id || isNaN(id)) {
    return res.status(400).json({ error: 'Invalid user ID' });
  }
  next();
};

// Apply to specific route
app.get('/users/:id', validateUser, (req, res) => {
  res.json({ userId: req.params.id });
});

// Apply to multiple routes using array
const checkAuth = (req, res, next) => {
  const token = req.headers.authorization;
  if (!token) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  next();
};

const checkRole = (req, res, next) => {
  // Check if user has admin role
  req.userRole = 'admin';
  next();
};

// Chain multiple middleware
app.put('/admin/users/:id', [checkAuth, checkRole, validateUser], (req, res) => {
  res.json({ message: 'User updated by admin' });
});

🟣 Route-Level Middleware Flow:

Request → checkAuth → checkRole → validateUser → Route Handler → Response

Each middleware can modify the request or end the response early.

Middleware Chaining

Express allows you to chain multiple middleware functions together for a single route.

// Middleware functions
const middleware1 = (req, res, next) => {
  console.log('Middleware 1');
  req.data = { step: 1 };
  next();
};

const middleware2 = (req, res, next) => {
  console.log('Middleware 2');
  req.data.step = 2;
  next();
};

const middleware3 = (req, res, next) => {
  console.log('Middleware 3');
  req.data.step = 3;
  next();
};

app.get('/chain', middleware1, middleware2, middleware3, (req, res) => {
  console.log('Final handler');
  res.json({
    message: 'Middleware chain complete',
    data: req.data
  });
});

// Output in console:
// Middleware 1
// Middleware 2
// Middleware 3
// Final handler

Error Handling Middleware

Express has a special type of middleware for error handling. Error-handling middleware takes four parameters: (err, req, res, next).

🟣 Error-handling middleware should be placed after all other app.use() and route definitions.
const app = express();

// Route that throws an error
app.get('/error', (req, res, next) => {
  const error = new Error('Something went wrong!');
  error.status = 500;
  next(error); // Pass error to error handler
});

// Route that triggers a validation error
app.post('/users', (req, res, next) => {
  if (!req.body.name) {
    const error = new Error('Name is required');
    error.status = 400;
    return next(error);
  }
  res.json({ message: 'User created' });
});

// Error handling middleware
app.use((err, req, res, next) => {
  console.error('Error:', err.message);
  console.error('Stack:', err.stack);
  const status = err.status || 500;
  res.status(status).json({
    error: {
      message: err.message,
      status: status
    }
  });
});

// 404 handler (catch-all)
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

Environment Variables in Node.js

Environment variables are key-value pairs that allow you to configure your application without hardcoding sensitive information like API keys, database passwords, or port numbers.

🟣 Environment variables are essential for keeping sensitive data secure and making your application configurable across different environments (development, staging, production).
// Accessing environment variables
const port = process.env.PORT || 3000;
const dbUrl = process.env.DATABASE_URL;
const apiKey = process.env.API_KEY;

console.log(`Server running on port ${port}`);
console.log(`Database URL: ${dbUrl}`);
$ PORT=5000 node app.js
Server running on port 5000

dotenv Package – Managing .env Files

dotenv is a zero-dependency module that loads environment variables from a .env file into process.env. This makes it easy to manage configuration in development.

// Install dotenv
npm install dotenv

// Create .env file
// .env
PORT=5000
DB_HOST=localhost
DB_USER=admin
DB_PASSWORD=secret123
JWT_SECRET=mySuperSecretKey
NODE_ENV=development

// app.js
require('dotenv').config();

console.log('Port:', process.env.PORT); // 5000
console.log('DB Host:', process.env.DB_HOST); // localhost
console.log('Environment:', process.env.NODE_ENV); // development

.env Configuration Best Practices

Follow these best practices when working with environment variables:

  • Never commit .env files – add .env to .gitignore.
  • Use .env.example – create a sample file with placeholder values for other developers.
  • Validate required variables – check if all required environment variables are set.
  • Use different .env files – .env.development, .env.production, etc.
  • Keep sensitive data out – never hardcode secrets in your source code.
  • Use environment-specific config – load different config based on NODE_ENV.
// .env.example
PORT=3000
DB_HOST=localhost
DB_USER=username
DB_PASSWORD=password
JWT_SECRET=your-secret-key
NODE_ENV=development

// .gitignore
node_modules/
.env
.env.*

// app.js - validation
const requiredEnv = ['PORT', 'DB_HOST', 'JWT_SECRET'];
requiredEnv.forEach(key => {
  if (!process.env[key]) {
    console.error(`Error: ${key} is not defined in environment variables`);
    process.exit(1);
  }
});

Complete Example: Middleware + Environment Variables

A full Express application combining all the concepts covered.

require('dotenv').config();
const express = require('express');
const app = express();

// --- Application-Level Middleware ---
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Logger middleware
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url} - ${new Date().toISOString()}`);
  next();
});

// --- Custom Middleware ---
const validateApiKey = (req, res, next) => {
  const apiKey = req.headers['x-api-key'];
  if (!apiKey || apiKey !== process.env.API_KEY) {
    return res.status(401).json({ error: 'Invalid API key' });
  }
  next();
};

const rateLimiter = (req, res, next) => {
  // Simple rate limiting - track requests per IP
  const ip = req.ip;
  if (!global.rateLimits) global.rateLimits = {};
  if (!global.rateLimits[ip]) {
    global.rateLimits[ip] = { count: 1, resetTime: Date.now() + 60000 };
  } else {
    const data = global.rateLimits[ip];
    if (Date.now() > data.resetTime) {
      data.count = 1;
      data.resetTime = Date.now() + 60000;
    } else if (data.count > 10) {
      return res.status(429).json({ error: 'Too many requests' });
    } else {
      data.count++;
    }
  }
  next();
};

// --- Route-Level Middleware ---
const checkAuth = (req, res, next) => {
  const token = req.headers.authorization;
  if (!token) {
    return res.status(401).json({ error: 'No token provided' });
  }
  // Simulate token verification
  if (token !== 'Bearer ' + process.env.JWT_SECRET) {
    return res.status(403).json({ error: 'Invalid token' });
  }
  req.user = { id: 1, role: 'admin' };
  next();
};

// --- Routes ---
app.get('/', (req, res) => {
  res.json({
    message: 'Hello World!',
    environment: process.env.NODE_ENV
  });
});

// Public route
app.get('/api/public', rateLimiter, (req, res) => {
  res.json({ message: 'Public data' });
});

// Protected route with middleware chain
app.get('/api/protected', rateLimiter, validateApiKey, checkAuth, (req, res) => {
  res.json({
    message: 'Protected data',
    user: req.user
  });
});

// Route with parameters
app.get('/api/users/:id', rateLimiter, (req, res, next) => {
  const id = parseInt(req.params.id);
  if (isNaN(id)) {
    const error = new Error('Invalid user ID');
    error.status = 400;
    return next(error);
  }
  res.json({ userId: id, name: `User ${id}` });
});

// --- Error Handling ---
app.use((err, req, res, next) => {
  console.error('Error:', err.message);
  const status = err.status || 500;
  res.status(status).json({
    error: {
      message: err.message,
      status: status
    }
  });
});

// 404 handler
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

// --- Start Server ---
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`🚀 Server running on port ${PORT}`);
  console.log(`📦 Environment: ${process.env.NODE_ENV || 'development'}`);
});
